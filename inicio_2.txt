# INICIAR MYSQL DESPUÉS DEL REINICIO
# ===================================

## COMANDOS:
ssh usuario@IP_DEL_SERVIDOR
cd /var/www/clinica-back

# Verificar archivos necesarios:
ls -la docker-compose.yml mysql.cnf

# Si faltan archivos, actualizar repositorio:
git pull origin master

# Si sigue sin funcionar, verificar contenido:
cat docker-compose.yml

docker compose up -d

## CONFIGURAR PERMISOS MYSQL:
# Esperar que MySQL inicie completamente
sleep 10

# Configurar permisos para acceso remoto:
docker compose exec mysql mysql -u root -proot2025 -e "GRANT ALL PRIVILEGES ON *.* TO 'root'@'%' WITH GRANT OPTION; FLUSH PRIVILEGES;"

## VERIFICAR:
docker compose ps
docker compose logs mysql

## CONECTAR A MYSQL:

# Desde el mismo servidor (local):
docker compose exec mysql mysql -u root -proot2025

# NOTA: mysql -h localhost NO funciona sin MySQL instalado en host

# Desde otra máquina (remoto):
mysql -h IP_DEL_SERVIDOR -u root -proot2025

## COMANDOS ÚTILES:
docker compose restart mysql
docker compose logs -f mysql
docker compose down

## TROUBLESHOOTING:
# Si error "no configuration file provided":
git status
git pull origin master
ls -la docker-compose.yml

# Si mysql.cnf no existe:
cat > mysql.cnf << EOF
[mysqld]
bind-address = 0.0.0.0
default-authentication-plugin = mysql_native_password
EOF

# SOLUCIÓN PARA ERROR DE PERMISOS:
# Si sale "Access denied for user 'root'@'172.19.0.X'":

# 1. Conectar a MySQL:
docker compose exec mysql mysql -u root -proot2025

# 2. Ejecutar dentro de MySQL:
# SELECT User, Host FROM mysql.user WHERE User='root';
# GRANT ALL PRIVILEGES ON *.* TO 'root'@'%' WITH GRANT OPTION;
# CREATE USER IF NOT EXISTS 'root'@'%' IDENTIFIED BY 'root2025';
# GRANT ALL PRIVILEGES ON *.* TO 'root'@'%' WITH GRANT OPTION;
# FLUSH PRIVILEGES;
# EXIT;

# 3. Verificar con Laravel:
# php artisan config:clear
# php artisan migrate:status
