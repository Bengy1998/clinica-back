# COMANDOS DOCKER LARAVEL - REFERENCIA RÁPIDA
# ============================================

## GESTIÓN DE CONTENEDORES
## ========================

# Construir e iniciar todos los servicios
docker-compose up -d --build

# Iniciar servicios (sin rebuild)
docker-compose up -d

# Detener todos los servicios
docker-compose down

# Reiniciar todos los servicios
docker-compose restart

# Reiniciar un servicio específico
docker-compose restart nginx
docker-compose restart php
docker-compose restart mysql

## VER LOGS
## ========

# Ver logs de todos los servicios
docker-compose logs -f

# Ver logs de un servicio específico
docker-compose logs -f nginx
docker-compose logs -f php
docker-compose logs -f mysql

# Ver últimas 50 líneas de logs
docker-compose logs --tail=50 nginx

## ACCEDER A CONTENEDORES
## ======================

# Acceder al contenedor PHP
docker-compose exec php sh

# Acceder a MySQL
docker-compose exec mysql mysql -u root -proot2025

## COMANDOS LARAVEL
## ================

# Ejecutar migraciones
docker-compose exec php php artisan migrate

# Ejecutar seeders
docker-compose exec php php artisan db:seed

# Limpiar caché
docker-compose exec php php artisan optimize:clear

# Instalar dependencias
docker-compose exec php composer install

# Generar clave de aplicación
docker-compose exec php php artisan key:generate

# Crear enlace de storage
docker-compose exec php php artisan storage:link

## GESTIÓN DE BASE DE DATOS
## ========================

# Backup de base de datos
docker-compose exec mysql mysqldump -u root -proot2025 clinica > backup_$(date +%Y%m%d_%H%M%S).sql

# Restaurar base de datos
docker-compose exec -T mysql mysql -u root -proot2025 clinica < backup.sql

# Conectar a MySQL desde host
mysql -h localhost -P 3306 -u root -proot2025

## COMANDOS MAKEFILE (SI DISPONIBLE)
## =================================

make help           # Mostrar ayuda
make up             # Iniciar servicios
make down           # Detener servicios
make restart        # Reiniciar servicios
make logs           # Ver logs
make shell          # Acceder a PHP shell
make mysql-shell    # Acceder a MySQL

## LIMPIEZA RÁPIDA
## ===============

# Limpiar Docker
docker system prune -f

# Reiniciar todo
docker-compose down && docker-compose up -d

## COMANDOS DE EMERGENCIA
## ======================

# Reiniciar servicios Ubuntu
sudo systemctl restart docker nginx

# Ver logs rápidos
docker-compose logs --tail=50
sudo tail -f /var/log/nginx/error.log

# Ver estado
docker-compose ps
sudo systemctl status docker nginx

# Test conectividad
curl -I http://localhost:8080    # Docker
curl -I https://tu-dominio.com   # Con SSL

# Verificar puertos
sudo ss -tlnp | grep -E ':80|:443|:8080|:3306'

## CONFIGURACIÓN SSL RÁPIDA
## =========================

# PASO 1: Cambiar puerto Docker de 80 a 8080
sed -i 's/"80:80"/"8080:80"/g' docker-compose.yml
docker-compose down && docker-compose up -d

# PASO 2: Instalar Nginx en Ubuntu
sudo apt install nginx certbot python3-certbot-nginx -y

# PASO 3: Usar script automático SSL
chmod +x setup_ssl.sh && sudo ./setup_ssl.sh
