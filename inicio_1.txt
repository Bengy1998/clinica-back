# INSTALACIÓN DOCKER + MYSQL
# ===========================

## INSTALAR DOCKER:
sudo apt update && sudo apt upgrade -y
sudo apt install apt-transport-https ca-certificates curl gnupg lsb-release -y
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | sudo gpg --dearmor -o /usr/share/keyrings/docker-archive-keyring.gpg
echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/docker-archive-keyring.gpg] https://download.docker.com/linux/ubuntu $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null
sudo apt update
sudo apt install docker-ce docker-ce-cli containerd.io docker-compose-plugin -y
sudo usermod -aG docker $USER

## VERIFICAR INSTALACIÓN:
# Después del reinicio, verificar con:
# docker --version
# docker compose version
# Si 'docker compose' no funciona, usar: sudo apt install docker-compose

## CLONAR PROYECTO:
sudo mkdir -p /var/www
cd /var/www
sudo git clone https://github.com/Bengy1998/clinica-back.git clinica-back
cd clinica-back
sudo chown -R $USER:$USER /var/www/clinica-back

## CONFIGURAR ACCESO REMOTO:
cat > mysql.cnf << EOF
[mysqld]
bind-address = 0.0.0.0
default-authentication-plugin = mysql_native_password
EOF

sudo reboot
